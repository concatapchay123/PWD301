Viết hàm decorator @require_role('admin') trong Flask kiểm tra claim trong JWT payload từ request header. 

# Bài làm của sinh viên: Nguyễn Minh Anh (MA-8392)
from functools import wraps
from flask import request, jsonify, g
def require_role(expected_role):
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            user = getattr(g, 'current_user', None)
            if not user or user.get('role') != expected_role:
                return jsonify({'error': 'Forbidden: Insufficient privileges'}), 403
            return f(*args, **kwargs)
        return decorated_function
    return decorator

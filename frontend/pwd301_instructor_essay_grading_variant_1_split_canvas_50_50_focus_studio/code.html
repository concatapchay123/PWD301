from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from jose import JWTError, jwt
import redis

# Khởi tạo kết nối Redis Blacklist
redis_client = redis.Redis(host="localhost", port=6379, db=0, decode_responses=True)
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")

async def get_current_active_user(token: str = Depends(oauth2_scheme)):
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Không thể xác thực thông tin đăng nhập hoặc Token đã bị vô hiệu",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        # Kiểm tra blacklist Redis trước khi decode payload
        if redis_client.get(f"blacklist:{token}"):
            raise credentials_exception
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        if username is None:
            raise credentials_exception
    except JWTError:
        raise credentials_exception
    return {"username": username, "role": "authenticated_student"}
